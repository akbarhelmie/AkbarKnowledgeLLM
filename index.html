                        nodeMesh.add(ringMesh);

        function applyEdgeHighlight(clickedEdgeData) { 

            isHighlightActive = true; 

            const sourceId = clickedEdgeData.source; 

            const targetId = clickedEdgeData.target;


            nodes.forEach((nodeMesh, nodeId) => {

                const userData = nodeMesh.userData;

                const nodeData = userData.originalData;

                let targetOpacity = FAINT_OPACITY;

                

                if (nodeId === sourceId || nodeId === targetId) {

                    targetOpacity = nodeData.id === "Akbar" ? AKBAR_NODE_OPACITY : HIGHLIGHT_OPACITY;

                    if (nodeData.id === "Akbar") { 

                        nodeMesh.material.color.setHex(HIGHLIGHT_NODE_COLOR); 

                        if(nodeMesh.material.emissive) nodeMesh.material.emissive.setHex(HIGHLIGHT_NODE_COLOR); 

                        nodeMesh.material.emissiveIntensity = 0.7; 

                    } else { 

                        nodeMesh.material.color.setHex(HIGHLIGHT_NODE_COLOR); 

                        if(nodeMesh.material.emissive) nodeMesh.material.emissive.setHex(HIGHLIGHT_NODE_COLOR); else if (nodeMesh.material.emissive !==undefined) nodeMesh.material.emissive = new THREE.Color(HIGHLIGHT_NODE_COLOR);

                         nodeMesh.material.emissiveIntensity = 0.5; 

                    }

                     (userData.orbitingRings || []).forEach(r => { r.mesh.material.opacity = NODE_RING_OPACITY + 0.25; r.mesh.material.needsUpdate = true;});

                } else { 

                    if (nodeData.id === "Akbar") { 

                        nodeMesh.material.color.setHex(0x5F9EA0); 

                        if(nodeMesh.material.emissive) nodeMesh.material.emissive.setHex(0x5F9EA0);

                        nodeMesh.material.emissiveIntensity = 0.15; 

                        targetOpacity = AKBAR_NODE_OPACITY * 0.6; 

                    } else { 

                        nodeMesh.material.color.setHex(nodeData.originalColor); 

                        if(nodeMesh.material.emissive) nodeMesh.material.emissive.setHex(0x000000); 

                    }

                    (userData.orbitingRings || []).forEach(r => { r.mesh.material.opacity = FAINT_OPACITY + 0.02; r.mesh.material.needsUpdate = true;});

                }

                nodeMesh.material.opacity = targetOpacity; 

                nodeMesh.material.needsUpdate = true;

            });


            edges.forEach(edgeWrapper => {

                if (!edgeWrapper || !edgeWrapper.dna || !edgeWrapper.originalData) return;

                const edge = edgeWrapper.originalData; const dna = edgeWrapper.dna;

                const isClickedEdge = edge.source === sourceId && edge.target === targetId && edge.relation === clickedEdgeData.relation;

                const isRelatedToHighlightedNodes = edge.source === sourceId || edge.target === sourceId || edge.source === targetId || edge.target === targetId;

                

                dna.isPulsing = false; 


                if (isClickedEdge) { 

                    dna.currentRadius = dna.highlightRadius; 

                    if(dna.strand1) {dna.strand1.material.color.setHex(HIGHLIGHT_EDGE_COLOR); dna.strand1.material.opacity = HIGHLIGHT_OPACITY * 0.7;}

                    if(dna.strand2) {dna.strand2.material.color.setHex(HIGHLIGHT_EDGE_COLOR); dna.strand2.material.opacity = HIGHLIGHT_OPACITY * 0.7;}

                } else if (isRelatedToHighlightedNodes) { 

                    dna.currentRadius = dna.normalRadius; 

                    if(dna.strand1) {dna.strand1.material.color.setHex(new THREE.Color(dna.originalColor1).lerp(new THREE.Color(HIGHLIGHT_EDGE_COLOR), 0.6).getHex()); dna.strand1.material.opacity = NORMAL_EDGE_OPACITY + 0.25; }

                    if(dna.strand2) {dna.strand2.material.color.setHex(new THREE.Color(dna.originalColor2).lerp(new THREE.Color(HIGHLIGHT_EDGE_COLOR), 0.6).getHex()); dna.strand2.material.opacity = NORMAL_EDGE_OPACITY + 0.25;}

                } else { 

                    dna.currentRadius = dna.normalRadius;

                    if(dna.strand1) {dna.strand1.material.color.setHex(dna.originalColor1); dna.strand1.material.opacity = FAINT_OPACITY * 0.6; }

                    if(dna.strand2) {dna.strand2.material.color.setHex(dna.originalColor2); dna.strand2.material.opacity = FAINT_OPACITY * 0.6;}

                }

                if(dna.strand1 && dna.strand1.material) dna.strand1.material.needsUpdate = true; 

                if(dna.strand2 && dna.strand2.material) dna.strand2.material.needsUpdate = true;

            });

        }

        

        function updateAllDNAStrands(currentTime) {

            if (!edges || edges.length === 0) return;

            edges.forEach(edgeWrapper => {

                if (!edgeWrapper || !edgeWrapper.dna || !edgeWrapper.originalData || !edgeWrapper.dna.strand1 || !edgeWrapper.dna.strand2) {return; }

                const dna = edgeWrapper.dna; 

                const sourceNode = nodes.get(edgeWrapper.originalData.source); 

                const targetNode = nodes.get(edgeWrapper.originalData.target);

                if (!sourceNode || !targetNode) { dna.strand1.visible = false; dna.strand2.visible = false; return; }

                

                const p1 = sourceNode.position; const p2 = targetNode.position; 

                dna.phase += 0.035; 

                

                if (!isHighlightActive) { 

                    if (dna.isPulsing) {

                        const pulseElapsedTime = currentTime - dna.pulseStartTime;

                        if (pulseElapsedTime < EDGE_PULSE_DURATION) {

                            const pulseProgress = pulseElapsedTime / EDGE_PULSE_DURATION;

                            const currentPulseOpacity = NORMAL_EDGE_OPACITY + (EDGE_PULSE_OPACITY_MAX - NORMAL_EDGE_OPACITY) * Math.sin(pulseProgress * Math.PI); 

                            dna.strand1.material.opacity = currentPulseOpacity;

                            dna.strand2.material.opacity = currentPulseOpacity;

                            const c1 = new THREE.Color(dna.originalColor1).multiplyScalar(1 + (EDGE_PULSE_COLOR_BRIGHTEN_FACTOR -1) * Math.sin(pulseProgress * Math.PI));

                            const c2 = new THREE.Color(dna.originalColor2).multiplyScalar(1 + (EDGE_PULSE_COLOR_BRIGHTEN_FACTOR -1) * Math.sin(pulseProgress * Math.PI));

                            dna.strand1.material.color.set(c1);

                            dna.strand2.material.color.set(c2);

                        } else {

                            dna.isPulsing = false;

                            dna.strand1.material.opacity = NORMAL_EDGE_OPACITY;

                            dna.strand2.material.opacity = NORMAL_EDGE_OPACITY;

                            dna.strand1.material.color.setHex(dna.originalColor1);

                            dna.strand2.material.color.setHex(dna.originalColor2);

                            dna.nextPulseTime = currentTime + EDGE_PULSE_INTERVAL_MIN + Math.random() * (EDGE_PULSE_INTERVAL_MAX - EDGE_PULSE_INTERVAL_MIN);

                        }

                    } else if (currentTime > dna.nextPulseTime) {

                        dna.isPulsing = true;

                        dna.pulseStartTime = currentTime;

                    }

                }


                const points1 = []; const points2 = []; const direction = new THREE.Vector3().subVectors(p2, p1); const edgeLength = direction.length();

                if (edgeLength < 0.01) { dna.strand1.visible = false; dna.strand2.visible = false; return; } 

                else { dna.strand1.visible = true; dna.strand2.visible = true; }

                direction.normalize(); let up = new THREE.Vector3(0, 1, 0); if (Math.abs(direction.y) > 0.99) { up.set(1, 0, 0); } 

                const ortho1 = new THREE.Vector3().crossVectors(direction, up).normalize(); const ortho2 = new THREE.Vector3().crossVectors(direction, ortho1).normalize();

                for (let i = 0; i <= dna.numSegments; i++) {

                    const t = i / dna.numSegments; const currentPointOnEdge = new THREE.Vector3().copy(p1).addScaledVector(direction, t * edgeLength);

                    const angle = t * dna.numTurns * 2 * Math.PI + dna.phase; const radius = dna.currentRadius; 

                    const offset1 = new THREE.Vector3().addScaledVector(ortho1, Math.cos(angle) * radius).addScaledVector(ortho2, Math.sin(angle) * radius); points1.push(currentPointOnEdge.clone().add(offset1));

                    const offset2 = new THREE.Vector3().addScaledVector(ortho1, Math.cos(angle + Math.PI) * radius).addScaledVector(ortho2, Math.sin(angle + Math.PI) * radius); points2.push(currentPointOnEdge.clone().add(offset2));

                }

                if (dna.strand1.geometry && dna.strand1.geometry.attributes.position) { dna.strand1.geometry.setFromPoints(points1); dna.strand1.geometry.attributes.position.needsUpdate = true; }

                if (dna.strand2.geometry && dna.strand2.geometry.attributes.position) { dna.strand2.geometry.setFromPoints(points2); dna.strand2.geometry.attributes.position.needsUpdate = true; }

                if (dna.strand1.material) dna.strand1.material.needsUpdate = true;

                if (dna.strand2.material) dna.strand2.material.needsUpdate = true;

            });

        }

        function updateNodeOrbitingRings() {

            nodes.forEach(nodeMesh => {

                if (nodeMesh.userData.orbitingRings && nodeMesh.userData.orbitingRings.length > 0) {

                    nodeMesh.userData.orbitingRings.forEach(ringData => {

                        // Rings are children, their rotation.x = PI/2 makes them flat in node's local XZ.

                        // We rotate their local Y axis (which is now perpendicular to their plane, effectively the node's original Y)

                        // to make them spin around the node.

                        ringData.mesh.rotation.y += ringData.speed; 

                    });

                }

            });

        }

        function updateNodeHeartbeats(deltaTime) {

            nodes.forEach(nodeMesh => {

                const userData = nodeMesh.userData;

                if (!userData || typeof userData.heartbeatPhase === 'undefined' || !userData.baseScaleVec) return; 


                const speed = (userData.originalData.id === "Akbar") ? NODE_HEARTBEAT_SPEED * AKBAR_HEARTBEAT_SPEED_MULTIPLIER : NODE_HEARTBEAT_SPEED;

                userData.heartbeatPhase += speed * deltaTime;

                if (userData.heartbeatPhase > Math.PI * 2) userData.heartbeatPhase -= Math.PI * 2;

                

                const scaleRange = NODE_HEARTBEAT_MAX_SCALE - NODE_HEARTBEAT_MIN_SCALE;

                const scaleFactor = NODE_HEARTBEAT_MIN_SCALE + (Math.sin(userData.heartbeatPhase) * 0.5 + 0.5) * scaleRange;

                nodeMesh.scale.set(userData.baseScaleVec.x * scaleFactor, userData.baseScaleVec.y * scaleFactor, userData.baseScaleVec.z * scaleFactor);

            });

        }


        let activeBlinks = new Map(); 

        function triggerNodeBlinks(clickedNodeId) {

            activeBlinks.clear(); 


            (graphDataFull.edges || []).forEach(edge => {

                let connectedNodeId = null;

                if (edge.source === clickedNodeId) connectedNodeId = edge.target;

                else if (edge.target === clickedNodeId) connectedNodeId = edge.source;


                if (connectedNodeId && connectedNodeId !== clickedNodeId) { 

                    const nodeMesh = nodes.get(connectedNodeId);

                    if (nodeMesh && nodeMesh.material) { 

                        let originalEmissiveHex = 0x000000;

                        let originalEmissiveIntensity = 0;

                        let hasEmissive = false;

                        if (nodeMesh.material.emissive) { 

                            hasEmissive = true;

                            originalEmissiveHex = nodeMesh.material.emissive.getHex();

                            originalEmissiveIntensity = nodeMesh.material.emissiveIntensity;

                        } else if (nodeMesh.material.color) { 

                             originalEmissiveHex = nodeMesh.material.color.getHex(); 

                        }


                        activeBlinks.set(connectedNodeId, {

                            endTime: Date.now() + NODE_BLINK_DURATION,

                            blinkCycleTime: Date.now(),

                            originalEmissiveHex: originalEmissiveHex,

                            originalEmissiveIntensity: originalEmissiveIntensity,

                            blinkCount: 0,

                            isLit: false,

                            hasEmissive: hasEmissive

                        });

                    }

                }

            });

             console.log("TRIGGER BLINKS: For " + activeBlinks.size + " nodes connected to " + clickedNodeId);

        }


        function updateActiveNodeBlinks(currentTime) {

            if (activeBlinks.size === 0) return;


            activeBlinks.forEach((blinkData, nodeId) => {

                const nodeMesh = nodes.get(nodeId);

                if (!nodeMesh || !nodeMesh.material) { activeBlinks.delete(nodeId); return; }


                if (currentTime > blinkData.endTime || blinkData.blinkCount >= NODE_BLINK_MAX_BLINKS * 2) { 

                    if(blinkData.hasEmissive) {

                        nodeMesh.material.emissive.setHex(blinkData.originalEmissiveHex);

                        nodeMesh.material.emissiveIntensity = blinkData.originalEmissiveIntensity;

                    } 

                    nodeMesh.material.needsUpdate = true;

                    activeBlinks.delete(nodeId);

                    return;

                }


                if (currentTime > blinkData.blinkCycleTime) {

                    blinkData.isLit = !blinkData.isLit;

                    if (blinkData.hasEmissive) {

                        if (blinkData.isLit) {

                            nodeMesh.material.emissive.setHex(0xffffff); 

                            nodeMesh.material.emissiveIntensity = 0.8; 

                        } else {

                            nodeMesh.material.emissive.setHex(blinkData.originalEmissiveHex);

                            nodeMesh.material.emissiveIntensity = blinkData.originalEmissiveIntensity;

                        }

                    } 

                    nodeMesh.material.needsUpdate = true;

                    blinkData.blinkCycleTime = currentTime + NODE_BLINK_INTERVAL;

                    blinkData.blinkCount++;

                }

            });

        }


        function onCanvasClick(event) { 

            const rect = renderer.domElement.getBoundingClientRect(); mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1; mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;

            raycaster.setFromCamera(mouse, camera); const intersects = raycaster.intersectObjects(Array.from(nodes.values())); 

            

            clearHighlights(); 

            if (hideEdgeDefinition) hideEdgeDefinition(); 

            if (hideNodeInspector) hideNodeInspector();


            if (intersects.length > 0) { 

                const clickedNodeMesh = intersects[0].object; 

                const nodeData = clickedNodeMesh.userData.originalData; 

                const nodeCategory = clickedNodeMesh.userData.category; 

                

                if(nodeConnectionsList) nodeConnectionsList.innerHTML = ''; let hasConnections = false; 

                if(nodeIdText) nodeIdText.textContent = (nodeData.displayText || nodeData.id).replace(/_/g, ' '); 

                if(nodeTypeText) nodeTypeText.textContent = nodeData.type.replace(/_/g, ' '); 

                if(nodeCategoryText) nodeCategoryText.textContent = nodeCategory || 'N/A'; 

                

                (graphDataFull.edges || []).forEach(edge => { let connectedNodeId = null, relationType = '', relationText = edge.relation ? edge.relation.replace(/_/g, ' ') : 'related to'; if (edge.source === nodeData.id) { connectedNodeId = edge.target; relationType = relationText + " &rarr;"; hasConnections = true; } else if (edge.target === nodeData.id) { connectedNodeId = edge.source; relationType = "&larr; " + relationText; hasConnections = true; } if (connectedNodeId) { const connectedNodeData = nodeDataMap.get(connectedNodeId); if(connectedNodeData && nodeConnectionsList){ const connectedNodeDisplay = (connectedNodeData.displayText || connectedNodeData.id).replace(/_/g, ' '); const listItem = document.createElement('li'); listItem.innerHTML = `${relationType} ${connectedNodeDisplay} (<em>${connectedNodeData.type.replace(/_/g, ' ')}</em>)`; nodeConnectionsList.appendChild(listItem); } } }); 

                if(nodeConnectionsDetails) nodeConnectionsDetails.style.display = hasConnections ? 'block' : 'none'; 

                if(nodeInspectorDisplay) nodeInspectorDisplay.style.display = 'block';

                briefInteractionPauseUntil = Date.now() + ROTATION_PAUSE_DURATION + NODE_BLINK_DURATION; 

            }

        }

        

        let lastFrameTime = 0;

        function animate(currentTime) { 

            requestAnimationFrame(animate); 

            if (!lastFrameTime && currentTime > 0) { lastFrameTime = currentTime; } 

            const deltaTime = (currentTime - lastFrameTime) * 0.001 || 0.0166; 

            lastFrameTime = currentTime;


            let canAutoRotate = true;

            if (isGraphDragging) { canAutoRotate = false; }

            if (Date.now() < briefInteractionPauseUntil) { canAutoRotate = false; }

            if (canAutoRotate && graphGroup) { graphGroup.rotation.y += autoRotationSpeed; }

            

            if (controls) controls.update(); 

            updateAllDNAStrands(currentTime); 

            updateNodeOrbitingRings();

            updateNodeHeartbeats(deltaTime); 

            updateActiveNodeBlinks(currentTime);


            nodes.forEach((nodeMesh) => {

                if (nodeMesh.userData.label && nodeMesh.position && nodeMesh.scale) { 

                    nodeMesh.userData.label.position.copy(nodeMesh.position); 

                    nodeMesh.userData.label.position.y += (3.0 * nodeMesh.scale.y); 

                }

            });

            edges.forEach(edgeWrapper => { 

                 if (edgeWrapper.label && edgeWrapper.originalData) {

                    const sourceNodeMesh = nodes.get(edgeWrapper.originalData.source); const targetNodeMesh = nodes.get(edgeWrapper.originalData.target);

                    if (sourceNodeMesh && targetNodeMesh && sourceNodeMesh.position && targetNodeMesh.position) { 

                        const midpoint = new THREE.Vector3().addVectors(sourceNodeMesh.position, targetNodeMesh.position).multiplyScalar(0.5); 

                        edgeWrapper.label.position.copy(midpoint); 

                    }

                 }

            });

            if(renderer && scene && camera) renderer.render(scene, camera); 

            if(labelRenderer && scene && camera) labelRenderer.render(scene, camera);

        }
